async function getWeather() { try { const response = await fetch('https://api.allorigins.win/get?url=https%3A%2F%2Fwttr.in/Moscow?lang=ru&format=json'); const jsonString = await response.json() const weatherData = JSON.parse(jsonString.contents); const tempC = weatherData.current_condition[0].temp_C; const humidity = weatherData.current_condition[0].humidity; const description = weatherData.current_condition[0].weatherDesc[0].value const htmlOutput = `

Температура: ${tempC}°C

Влажность: ${humidity}%

Погода: ${description}

`; document.getElementById("weather").innerHTML = htmlOutput } catch (error) { console.error(error.message); alert('Не удалось загрузить погоду.'); } }
